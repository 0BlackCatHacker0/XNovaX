import os, json, subprocess, sys, textwrap

ROOT=r"C:\NovaX"
CORE=os.path.join(ROOT,"core")
MEM=os.path.join(ROOT,"memory")
WEB=os.path.join(ROOT,"web")

def w(p,c):
    os.makedirs(os.path.dirname(p),exist_ok=True)
    open(p,"w",encoding="utf-8").write(textwrap.dedent(c))

print("Creating directories...")
for d in [ROOT,CORE,MEM,WEB]:
    os.makedirs(d,exist_ok=True)

# ---------------- DEPENDENCIES ----------------
print("Installing dependencies...")
subprocess.call([sys.executable,"-m","pip","install",
"flask","flask-socketio","eventlet","requests","psutil","pywin32"])

# ---------------- MEMORY FILES ----------------
for f in ["skills.json","projects.json","ai_routing.json","ui_routines.json","session_capture.json"]:
    p=os.path.join(MEM,f)
    if not os.path.exists(p): open(p,"w").write("{}" if "session" not in f else "[]")

# ---------------- TASK CLASSIFIER ----------------
w(CORE+r"\task_classifier.py",'''
def classify(text):
    t=text.lower()
    if any(x in t for x in ["code","bug","function","script"]): return "coding"
    if any(x in t for x in ["install","setup","configure"]): return "system"
    if any(x in t for x in ["design","idea","creative"]): return "creative"
    if any(x in t for x in ["explain","research","analyze"]): return "analysis"
    return "general"
''')

# ---------------- AI ROUTER MEMORY ----------------
w(CORE+r"\ai_router_memory.py",'''
import json, os
FILE=r"C:\\NovaX\\memory\\ai_routing.json"

def load():
    if not os.path.exists(FILE): return {}
    return json.load(open(FILE))

def save(d): json.dump(d,open(FILE,"w"),indent=2)

def record(task,model,success,time):
    d=load()
    d.setdefault(task,{})
    d[task].setdefault(model,{"runs":0,"success":0,"time":0})
    m=d[task][model]
    m["runs"]+=1
    m["success"]+=1 if success else 0
    m["time"]+=time
    save(d)

def best(task):
    d=load().get(task)
    if not d: return None
    return max(d,key=lambda k:(d[k]["success"]/(d[k]["runs"]+1)))
''')

# ---------------- SERVER ----------------
w(CORE+r"\server.py",'''
from flask import Flask,request,jsonify,send_from_directory
from flask_socketio import SocketIO
import subprocess, time, sys
from task_classifier import classify
from ai_router_memory import best,record

app=Flask(__name__,static_folder="../web")
socketio=SocketIO(app,cors_allowed_origins="*")

@app.route("/")
def index():
    return send_from_directory("../web","index.html")

@app.post("/think")
def think():
    text=request.json["text"]
    t=classify(text)
    model=best(t) or "default"
    return jsonify({"reply":f"Routing to {model}","model":model})

@app.post("/execute")
def execute():
    cmd=request.json["cmd"]
    start=time.time()
    try:
        out=subprocess.check_output(cmd,shell=True,stderr=subprocess.STDOUT,timeout=60).decode()
        record("system","local",True,time.time()-start)
        return jsonify({"ok":True,"out":out})
    except Exception as e:
        record("system","local",False,time.time()-start)
        return jsonify({"ok":False,"out":str(e)})

if __name__=="__main__":
    socketio.run(app,port=9999)
''')
