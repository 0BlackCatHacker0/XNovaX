from task_classifier import classify
from ai_router_memory import best

TABLE={
 "system":"executor",
 "coding":"deepseek",
 "analysis":"claude",
 "creative":"claude",
 "vision":"llava",
 "fast":"phi3",
 "research":"claude",
 "general":"mixtral"
}

FALLBACK={
 "deepseek":"claude",
 "claude":"mixtral",
 "mixtral":"phi3"
}

def choose(text):

    category=classify(text)

    # learned override
    learned=best(category)
    if learned:
        return learned,category,"learned"

    primary=TABLE.get(category,"mixtral")
    return primary,category,"rule"

task_classifier.py
def classify(text):
    t=text.lower()

    if any(x in t for x in ["install","open","delete","create file","run","start","stop"]):
        return "system"

    if any(x in t for x in ["error","bug","exception","traceback","fix","refactor"]):
        return "coding"

    if any(x in t for x in ["architecture","design a system","plan a program"]):
        return "analysis"

    if any(x in t for x in ["story","idea","brainstorm","creative","name"]):
        return "creative"

    if any(x in t for x in ["screenshot","image","see screen"]):
        return "vision"

    if len(t)<40:
        return "fast"

    if len(t)>400:
        return "research"

    return "analysis"

from router_engine import choose
from claude_connector import ask_claude

@app.post("/think")
def think():
    text=request.json["text"]

    model,category,mode=choose(text)

    if model=="claude":
        reply=ask_claude(text)
    else:
        reply=f"[{model} handling {category}]"

    return jsonify({
        "reply":reply,
        "model":model,
        "category":category,
        "mode":mode
    })
